```python
"""Tests for DIAG-04: Regime-conditional forward return analysis.

Tests compute_regime_forward_returns() in src/core/evaluation.py.
These are diagnostic validity tests — forward returns are NEVER model features.
"""

import numpy as np
import pandas as pd
import pytest
from src.core.evaluation import compute_regime_forward_returns


def _make_synthetic_data(n=500, seed=42, separated=True):
    """Create synthetic results + market DataFrames for testing.

    If separated=True: regimes have clearly different SPY forward returns.
    If separated=False: regimes have identical returns (KW should fail to reject H0).
    """
    rng = np.random.RandomState(seed)
    dates = pd.date_range('2015-01-01', periods=n, freq='B')

    # Assign regimes in blocks to simulate realistic dwell times
    regime_seq = []
    for i in range(n):
        block = i // 30
        regime_seq.append(['Low-Vol', 'Med-Vol', 'High-Vol'][block % 3])
    regimes = pd.Series(regime_seq, index=dates, name='regime_name')
    results = pd.DataFrame({'regime_name': regimes})

    # Generate market prices
    if separated:
        # Different drift per regime
        regime_drift = {'Low-Vol': 0.0008, 'Med-Vol': 0.0002, 'High-Vol': -0.0006}
    else:
        regime_drift = {'Low-Vol': 0.0004, 'Med-Vol': 0.0004, 'High-Vol': 0.0004}

    daily_returns = np.array([regime_drift[r] + rng.normal(0, 0.01) for r in regime_seq])

    spy_prices = 100 * np.exp(np.cumsum(daily_returns))
    eem_prices = 50 * np.exp(np.cumsum(daily_returns * 1.1 + rng.normal(0, 0.002, n)))
    tlt_prices = 120 * np.exp(np.cumsum(-daily_returns * 0.5 + rng.normal(0, 0.003, n)))
    hyg_prices = 80 * np.exp(np.cumsum(daily_returns * 0.7 + rng.normal(0, 0.002, n)))

    market = pd.DataFrame({
        'SPY_close': spy_prices,
        'EEM_close': eem_prices,
        'TLT_close': tlt_prices,
        'HYG_close': hyg_prices,
    }, index=dates)

    return results, market


class TestComputeRegimeForwardReturns:

    def test_returns_dataframe(self):
        results, market = _make_synthetic_data()
        df = compute_regime_forward_returns(results, market)
        assert isinstance(df, pd.DataFrame), "Must return a DataFrame"

    def test_required_columns(self):
        results, market = _make_synthetic_data()
        df = compute_regime_forward_returns(results, market)
        required = {'asset', 'horizon', 'regime', 'mean_return', 'n_obs', 'kw_pvalue'}
        assert required.issubset(set(df.columns)), f"Missing columns: {required - set(df.columns)}"

    def test_covers_all_assets_and_horizons(self):
        results, market = _make_synthetic_data()
        df = compute_regime_forward_returns(results, market)
        # Each asset+horizon combo must have at least one row
        expected_assets = ['SPY', 'EEM', 'TLT', 'HYG']
        expected_horizons = [1, 5, 21]
        for asset in expected_assets:
            for horizon in expected_horizons:
                subset = df[(df['asset'] == asset) & (df['horizon'] == horizon)]
                assert len(subset) > 0, f"No rows for asset={asset}, horizon={horizon}"

    def test_kw_pvalue_low_when_regimes_separated(self):
        """When regimes have clearly different returns, KW should reject H0."""
        results, market = _make_synthetic_data(n=600, separated=True)
        df = compute_regime_forward_returns(results, market)
        spy_1d = df[(df['asset'] == 'SPY') & (df['horizon'] == 1)]
        # kw_pvalue should be the same value for all rows of same asset/horizon
        # (one test per asset/horizon combo)
        kw_pvals = spy_1d['kw_pvalue'].dropna().unique()
        assert len(kw_pvals) > 0, "No KW p-values computed for SPY 1d"
        # At least one pvalue should show separation with n=600 clearly-different regimes
        assert any(p < 0.10 for p in kw_pvals), f"Expected low KW p-value for separated regimes, got {kw_pvals}"

    def test_kw_pvalue_high_when_regimes_identical(self):
        """When regimes have same return distribution, KW should not reject H0."""
        results, market = _make_synthetic_data(n=600, separated=False)
        df = compute_regime_forward_returns(results, market)
        spy_1d = df[(df['asset'] == 'SPY') & (df['horizon'] == 1)]
        kw_pvals = spy_1d['kw_pvalue'].dropna().unique()
        # Not guaranteed statistically but with same drift p should be > 0.05
        # Use a loose threshold to avoid flaky tests
        assert any(p > 0.01 for p in kw_pvals), f"Expected high KW p-value for identical regimes"

    def test_handles_missing_asset_column(self):
        """Should skip missing asset columns without raising."""
        results, market = _make_synthetic_data()
        market_partial = market.drop(columns=['EEM_close'])
        # Should not raise
        df = compute_regime_forward_returns(results, market_partial)
        # EEM rows should be absent
        assert len(df[df['asset'] == 'EEM']) == 0, "EEM should be absent when column missing"

    def test_kw_pvalue_in_valid_range(self):
        """All non-NaN p-values must be in [0, 1]."""
        results, market = _make_synthetic_data()
        df = compute_regime_forward_returns(results, market)
        pvals = df['kw_pvalue'].dropna()
        assert ((pvals >= 0) & (pvals <= 1)).all(), "p-values must be in [0, 1]"

    def test_n_obs_positive(self):
        """n_obs must be positive for all rows."""
        results, market = _make_synthetic_data()
        df = compute_regime_forward_returns(results, market)
        assert (df['n_obs'] > 0).all(), "n_obs must be positive"

    def test_returns_empty_dataframe_on_none_inputs(self):
        """Graceful handling of None inputs."""
        df = compute_regime_forward_returns(None, None)
        assert isinstance(df, pd.DataFrame), "Must return DataFrame even for None inputs"
        assert len(df) == 0, "Must return empty DataFrame for None inputs"
```

```python
def compute_regime_forward_returns(results, market,
                                   assets=('SPY', 'EEM', 'TLT', 'HYG'),
                                   horizons=(1, 5, 21)):
    """DIAG-04: Regime-conditional forward return analysis with Kruskal-Wallis test.

    For each (asset, horizon) combination, compute forward log-returns grouped by
    detected regime and test statistical separation using Kruskal-Wallis.

    **IMPORTANT: Diagnostic only.** Forward returns are NEVER fed back into the model
    as features. See REQUIREMENTS.md §Out of Scope for enforcement.

    Parameters
    ----------
    results : DataFrame or None
        Regime results with column 'regime_name' (str). DatetimeIndex.
    market : DataFrame or None
        Market data with columns '{ASSET}_close' for each asset. DatetimeIndex.
    assets : tuple of str
        Asset tickers to analyze. Default: ('SPY', 'EEM', 'TLT', 'HYG').
    horizons : tuple of int
        Forward return horizons in trading days. Default: (1, 5, 21).

    Returns
    -------
    df : DataFrame
        Columns: asset, horizon, regime, mean_return, median_return, std_return,
                 n_obs, kw_stat, kw_pvalue
        One row per (asset, horizon, regime) combination.
        kw_stat and kw_pvalue are the same within each (asset, horizon) group
        (one KW test per group, p-value broadcast to all regime rows in that group).
        Returns empty DataFrame if inputs are None or data is insufficient.

    Notes
    -----
    - Forward returns are computed as log(price[t+h] / price[t]) — forward-looking
      by construction. This is intentional: they measure what happened AFTER each
      detected regime, not what caused the detection.
    - KW test requires ≥ 2 regimes with ≥ 5 observations each. If not met,
      kw_stat and kw_pvalue are NaN for that (asset, horizon) group.
    - p-values are reported raw — no auto-flag or threshold enforcement (per D-07).
    """
    if results is None or market is None:
        return pd.DataFrame()

    regime_col = 'regime_name' if 'regime_name' in results.columns else results.columns[0]
    regimes = results[regime_col].dropna()

    rows = []

    for asset in assets:
        col = f"{asset}_close"
        if col not in market.columns:
            logger.debug("Asset column %s not found in market data — skipping %s", col, asset)
            continue

        prices = market[col].dropna()

        for h in horizons:
            # Forward log-return: shift prices back by h to align with regime date
            # fwd_ret[t] = log(price[t+h] / price[t])
            fwd_ret = np.log(prices.shift(-h) / prices).dropna()

            # Align with regime labels
            common_idx = regimes.index.intersection(fwd_ret.index)
            if len(common_idx) < 10:
                continue

            reg_aligned = regimes.loc[common_idx]
            ret_aligned = fwd_ret.loc[common_idx]

            # Compute per-regime statistics
            unique_regimes = sorted(reg_aligned.unique())
            groups = []
            regime_stats = {}

            for reg in unique_regimes:
                mask = reg_aligned == reg
                r = ret_aligned[mask].values
                if len(r) >= 5:  # Minimum observations for meaningful statistics
                    groups.append(r)
                    regime_stats[reg] = {
                        'mean_return': float(np.mean(r)),
                        'median_return': float(np.median(r)),
                        'std_return': float(np.std(r)),
                        'n_obs': int(len(r)),
                    }

            # Kruskal-Wallis test across all qualifying regimes
            kw_stat = np.nan
            kw_pvalue = np.nan
            if len(groups) >= 2:
                try:
                    kw_stat, kw_pvalue = kruskal(*groups)
                except Exception as e:
                    logger.debug("KW test failed for %s h=%d: %s", asset, h, e)

            # Build rows — one per (asset, horizon, regime)
            for reg, stats in regime_stats.items():
                rows.append({
                    'asset': asset,
                    'horizon': h,
                    'regime': reg,
                    'mean_return': stats['mean_return'],
                    'median_return': stats['median_return'],
                    'std_return': stats['std_return'],
                    'n_obs': stats['n_obs'],
                    'kw_stat': float(kw_stat) if not np.isnan(kw_stat) else np.nan,
                    'kw_pvalue': float(kw_pvalue) if not np.isnan(kw_pvalue) else np.nan,
                })

    return pd.DataFrame(rows) if rows else pd.DataFrame(
        columns=['asset', 'horizon', 'regime', 'mean_return', 'median_return',
                 'std_return', 'n_obs', 'kw_stat', 'kw_pvalue']
    )
```
